    <script>

        /* =================================
           VARIABLES
        ================================== */

        let saldoActual = 0;

        let movimientos = [];


        /* =================================
           ACTUALIZAR SALDO
        ================================== */

        function actualizarSaldo() {

            document.getElementById("saldo").textContent =
                "$" +
                saldoActual.toLocaleString("es-CO");

        }


        /* =================================
           CONSULTAR SALDO
        ================================== */

        function consultarSaldo() {

            alert(
                "Tu saldo actual es: $" +
                saldoActual.toLocaleString("es-CO")
            );

        }


        /* =================================
           MOSTRAR MOVIMIENTOS
        ================================== */

        function verMovimientos() {

            const seccion =
                document.getElementById(
                    "seccionMovimientos"
                );

            const lista =
                document.getElementById(
                    "listaMovimientos"
                );


            lista.innerHTML = "";


            if (movimientos.length === 0) {

                lista.innerHTML = `
                    <div class="sin-movimientos">
                        Todavía no tienes movimientos.
                    </div>
                `;

            } else {

                movimientos.forEach(
                    function(movimiento) {

                        const elemento =
                            document.createElement("div");

                        elemento.className =
                            "movimiento";

                        elemento.innerHTML = `

                            <div class="movimiento-info">

                                <strong>
                                    ${movimiento.tipo}
                                </strong>

                                <span>
                                    ${movimiento.fecha}
                                </span>

                            </div>

                            <div class="movimiento-valor">
                                +$${movimiento.valor.toLocaleString("es-CO")}
                            </div>

                        `;

                        lista.appendChild(elemento);

                    }
                );

            }


            seccion.style.display = "block";


            seccion.scrollIntoView({
                behavior: "smooth",
                block: "start"
            });

        }


        /* =================================
           OCULTAR MOVIMIENTOS
        ================================== */

        function ocultarMovimientos() {

            document.getElementById(
                "seccionMovimientos"
            ).style.display = "none";

        }


        /* =================================
           ABRIR RECARGA
        ================================== */

        function abrirRecarga() {

            document.getElementById(
                "modalRecarga"
            ).style.display = "flex";


            document.getElementById(
                "valorRecarga"
            ).value = "";


            document.getElementById(
                "valorRecarga"
            ).focus();

        }


        /* =================================
           CERRAR RECARGA
        ================================== */

        function cerrarRecarga() {

            document.getElementById(
                "modalRecarga"
            ).style.display = "none";

        }


        /* =================================
           CONFIRMAR RECARGA
        ================================== */

        function confirmarRecarga() {

            let valor =
                Number(
                    document.getElementById(
                        "valorRecarga"
                    ).value
                );


            if (!valor || valor <= 0) {

                alert(
                    "Por favor ingresa un valor válido."
                );

                return;

            }


            saldoActual += valor;


            const fecha =
                new Date().toLocaleString(
                    "es-CO",
                    {
                        dateStyle: "short",
                        timeStyle: "short"
                    }
                );


            movimientos.unshift({

                tipo: "Recarga",

                valor: valor,

                fecha: fecha

            });


            actualizarSaldo();


            cerrarRecarga();


            alert(
                "Recarga realizada correctamente.\n\n" +
                "Nuevo saldo: $" +
                saldoActual.toLocaleString("es-CO")
            );

        }


        /* =================================
           NFC
        ================================== */

        async function leerNFC() {

            if (!("NDEFReader" in window)) {

                alert(
                    "Tu navegador no permite leer tarjetas NFC mediante esta página."
                );

                return;

            }


            try {

                const ndef =
                    new NDEFReader();


                await ndef.scan();


                alert(
                    "Acerca la tarjeta NFC al teléfono."
                );


                ndef.onreading = event => {

                    alert(
                        "¡Tarjeta NFC detectada!"
                    );


                    console.log(
                        "Tarjeta NFC detectada:",
                        event.serialNumber
                    );

                };


            } catch (error) {

                console.error(error);


                alert(
                    "No fue posible iniciar la lectura NFC."
                );

            }

        }


        /* =================================
           INICIO
        ================================== */

        actualizarSaldo();

    </script>
